struct Donor
{
int id;
char name[30];
int age;
float weight;
char bloodGroup[5];
int days;
};

int checkEligibility(struct Donor d)
{
if (d.age >= 18 && d.age <= 65 &&
d.weight >= 50 && d.days >= 90)
return 1;
else
return 0;
}

{
printf("Enter Donor ID: ");

void findDonors(struct Donor d[], int n, char group[])
{
int i, found = 0;

printf("\nEligible Donors:\n");

for (i = 0; i < n; i++)
{
if (strcmp(d[i].bloodGroup, group) == 0 &&
checkEligibility(d[i]))
{
printf("ID: %d Name: %s\n",
d[i].id, d[i].name);
found = 1;
}
}

if (found == 0)
printf("No eligible donors found.\n");
}

void updateStock(int *stock, int units)
{
*stock = *stock + units;
printf("Stock updated successfully!\n");
}

{
FILE *fp = fopen("donors.txt", "w");

{
printf("File cannot be opened.\n");
return;

fclose(fp);
printf("Donor details saved successfully!\n");
}

struct Result
{
int count;
int youngestID;
};

struct Result getSummary(struct Donor d[], int n)
{
struct Result r;
int i, youngestAge = 999;

r.count = 0;
r.youngestID = -1;

for (i = 0; i < n; i++)
{
if (checkEligibility(d[i]))
{
r.count++;

if (d[i].age < youngestAge)
{
youngestAge = d[i].age;
r.youngestID = d[i].id;
}
}
}

return r;
}

/* Function to calculate digital root */
int digitalRoot(int n)
{
if (n < 10)
return n;

return digitalRoot(n % 10 + digitalRoot(n / 10));
}

/* Power using linear recursion */
int powerLinear(int base, int n)
{
if (n == 0)
return 1;

return base * powerLinear(base, n - 1);
}

/* Power using halving recursion */
int powerHalving(int base, int n)
{
int result;

if (n == 0)
return 1;

result = powerHalving(base, n / 2);

if (n % 2 == 0)
return result * result;
else
return base * result * result;
}

int main()
{
int id, base, n;

printf("===== BLOOD BANK RECURSION =====\n");

printf("Enter Donor ID: ");

{
printf("Please enter non-negative values.\n");
return 0;
}

printf("\nDigital Root = %d\n", digitalRoot(id));

printf("Power using Linear Recursion = %d\n",
powerLinear(base, n));

/* Recursive digital root */
int digital_root(int n) {
if (n < 10)
return n;

int sum = 0;

while (n > 0) {
sum += n % 10;
n = n / 10;
}

return digital_root(sum);
}

/* Linear recursive power */
long long power_linear(long long base, int exp) {
calls++;

if (exp == 0)
return 1;

return base * power_linear(base, exp - 1);
}

/* Fast recursive power */
long long power_fast(long long base, int exp) {
calls++;

if (exp == 0)
return 1;

long long half = power_fast(base, exp / 2);

if (exp % 2 == 0)
return half * half;

return base * half * half;
}

/* Binary stock file and update using fseek */
void stock_update() {
FILE *fp = fopen("stock.dat", "wb");

if (fp == NULL) {
perror("Stock file");
return;
}

struct Stock stock[5] = {
{1, 40}, {2, 25}, {3, 12}, {4, 8}, {5, 30}
};

fwrite(stock, sizeof(struct Stock), 5, fp);
fclose(fp);

int index, units;

printf("Enter record number to update (1-5): ");
scanf("%d", &index);

printf("Enter new units: ");
scanf("%d", &units);

fp = fopen("stock.dat", "rb+");

if (fp == NULL) {
perror("Stock file");
return;
}

struct Stock s;

if (index < 1 || index > 5) {
printf("Invalid record number!\n");
fclose(fp);
return;
}

if (fseek(fp, (index - 1) * sizeof(struct Stock),
SEEK_SET) != 0 ||
fread(&s, sizeof(struct Stock), 1, fp) != 1) {
printf("Record reading failed!\n");
fclose(fp);
return;
}

s.units = units;

if (fseek(fp, -(long)sizeof(struct Stock), SEEK_CUR) != 0 ||
fwrite(&s, sizeof(struct Stock), 1, fp) != 1) {
printf("Record updating failed!\n");
fclose(fp);
return;
}

printf("Stock updated successfully!\n");

rewind(fp);

printf("\n--- BLOOD STOCK ---\n");

while (fread(&s, sizeof(struct Stock), 1, fp) == 1) {
printf("Group code: %d Units: %d\n",
s.group_code, s.units);
}

fclose(fp);
}
